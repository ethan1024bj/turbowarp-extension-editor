1. 扩展基本结构

(function(Scratch) {
    'use strict';

    class MyExtension {
        getInfo() {
            return {
                id: 'myextension',      // 小写字母+数字
                name: 'My Extension',   // 显示名称
                color1: '#4C97FF',      // 主体颜色
                color2: '#3373CC',      // 内部颜色
                color3: '#295FA8',      // 深色
                blocks: [ ... ],        // 积木数组
                menus: { ... }          // 菜单定义（可选）
            };
        }
        // 积木方法写在这里
    }

    Scratch.extensions.register(new MyExtension());
})(Scratch);

5. 显示文本 (text)

用 [参数名] 插入参数占位符：

text: '将 [A] 和 [B] 相加'          // 两个参数
text: '说 [MESSAGE]'               // 一个参数
text: '当 [FLAG] 被点击时'          // 中文文本
text: 'set [VAR] to [VALUE]'       // 英文文本

5.5 积木附加属性

{
    opcode: 'myBlock',
    blockType: Scratch.BlockType.COMMAND,
    text: '...',
    hide: true,           // 隐藏积木（不在积木栏显示）
    isTerminal: true,     // 终端块（停止后续脚本）
    isEdgeActivated: true // 边缘触发（仅状态变化时触发一次）
}

6.5. 菜单选项定义示例

在 getInfo() 的 menus 中定义菜单，参数通过 menu 属性关联（参数类型必须为 STRING）：

getInfo() {
    return {
        // ...
        blocks: [
            {
                opcode: 'setColor',
                blockType: Scratch.BlockType.COMMAND,
                text: '设置颜色为 [COLOR]',
                arguments: {
                    COLOR: {
                        type: Scratch.ArgumentType.STRING,
                        menu: 'colorMenu'
                    }
                }
            }
        ],
        menus: {
            // 文字/值分离菜单
            colorMenu: {
                acceptReporters: true,
                items: [
                    { text: '红色', value: 'red' },
                    { text: '绿色', value: 'green' }
                ]
            },
            // 简写字符串列表
            simpleMenu: {
                acceptReporters: false,
                items: ['选项A', '选项B', '选项C']
            },
            // 带分隔线
            splitMenu: {
                items: ['功能1', '-', '功能2', '-', '功能3']
            },
            // 动态菜单（方法名）
            dynamicMenu: {
                acceptReporters: true,
                items: 'getDynamicItems'
            }
        }
    };
}

7. 函数实现 (方法体)

// COMMAND 指令块 - 无返回值
greet({ NAME }) {
    console.log('Hello ' + NAME);
}

// REPORTER 返回值块 - 必须 return
add({ A, B }) {
    return Number(A) + Number(B);
}

// BOOLEAN 布尔块 - 必须 return true/false
isEqual({ A, B }) {
    return String(A) === String(B);
}

// 使用 util 控制 C-block 执行次数
repeatN({ TIMES }, util) {
    for (let i = 0; i < TIMES; i++) {
        util.stackFrame.executed++;
        if (util.stackFrame.executed > TIMES) break;
    }
}

9. 完整示例

(function(Scratch) {
    'use strict';

    class MathTools {
        getInfo() {
            return {
                id: 'mathtools',
                name: '数学工具',
                color1: '#FF8C1A',
                color2: '#CF7013',
                color3: '#A8590F',
                blocks: [
                    {
                        opcode: 'add',
                        blockType: Scratch.BlockType.REPORTER,
                        text: '[A] + [B]',
                        arguments: {
                            A: { type: Scratch.ArgumentType.NUMBER, defaultValue: '0' },
                            B: { type: Scratch.ArgumentType.NUMBER, defaultValue: '0' }
                        }
                    },
                    {
                        opcode: 'pickColor',
                        blockType: Scratch.BlockType.REPORTER,
                        text: '选择颜色 [COLOR]',
                        arguments: {
                            COLOR: {
                                type: Scratch.ArgumentType.STRING,
                                menu: 'colorMenu'
                            }
                        }
                    }
                ],
                menus: {
                    colorMenu: {
                        acceptReporters: true,
                        items: [
                            { text: '红色', value: 'red' },
                            { text: '绿色', value: 'green' }
                        ]
                    }
                }
            };
        }

        add({ A, B }) {
            return Number(A) + Number(B);
        }

        pickColor({ COLOR }) {
            return COLOR;
        }
    }

    Scratch.extensions.register(new MathTools());
})(Scratch);

10. 加载扩展到 TurboWarp

方法一： 打开 turbowarp.org/editor → 左下角「高级」→ 「加载未打包的扩展」→ 粘贴代码或选择 .js 文件
方法二： 将 .js 文件上传到可访问的 URL，在扩展URL栏填入该地址
方法三： 在编辑器地址栏后加 #extension=URL 参数自动加载
